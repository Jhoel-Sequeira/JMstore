<span>Imprimir Arqueo (Z) y Cerrar Turno Definitivo [F10]</span>
</button>
</div>
</div>
</div>
</div>
<script>
  (function() {
    const cashInput = document.getElementById('input-cash');
    const cardsInput = document.getElementById('input-cards');
    const transfersInput = document.getElementById('input-transfers');
    
    const diffCash = document.getElementById('diff-cash');
    const diffCards = document.getElementById('diff-cards');
    const diffTransfers = document.getElementById('diff-transfers');

    const badgeCash = document.getElementById('badge-cash');
    const badgeCards = document.getElementById('badge-cards');
    const badgeTransfers = document.getElementById('badge-transfers');

    const totalCounted = document.getElementById('total-counted');
    const totalDiff = document.getElementById('total-diff');
    const overallStatus = document.getElementById('overall-status');

    const expectedCash = 1240.00;
    const expectedCards = 2860.00;
    const expectedTransfers = 750.00;
    const totalExpected = expectedCash + expectedCards + expectedTransfers;

    function formatCurrency(amount) {
      return '$' + Math.abs(amount).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }

    function updateRowState(inputEl, diffEl, badgeEl, expected) {
      const val = parseFloat(inputEl.value) || 0;
      const difference = val - expected;

      if (Math.abs(difference) < 0.01) {
        diffEl.textContent = '$0.00';
        diffEl.className = 'font-label-lg text-label-lg text-secondary font-mono';
        badgeEl.className = 'inline-flex items-center gap-1 px-2.5 py-1 rounded-full bg-secondary-fixed/50 text-on-secondary-fixed-variant font-label-sm text-label-sm';
        badgeEl.innerHTML = '<span class="w-1.5 h-1.5 rounded-full bg-secondary"></span>Cuadrado Exacto';
      } else if (difference > 0) {
        diffEl.textContent = '+' + formatCurrency(difference);
        diffEl.className = 'font-label-lg text-label-lg text-tertiary font-mono';
        badgeEl.className = 'inline-flex items-center gap-1 px-2.5 py-1 rounded-full bg-tertiary-fixed text-on-tertiary-fixed font-label-sm text-label-sm';
        badgeEl.innerHTML = '<span class="w-1.5 h-1.5 rounded-full bg-tertiary"></span>Sobrante';
      } else {
        diffEl.textContent = '-' + formatCurrency(difference);
        diffEl.className = 'font-label-lg text-label-lg text-error font-mono';
        badgeEl.className = 'inline-flex items-center gap-1 px-2.5 py-1 rounded-full bg-error-container text-on-error-container font-label-sm text-label-sm';
        badgeEl.innerHTML = '<span class="w-1.5 h-1.5 rounded-full bg-error"></span>Faltante';
      }
      return val;
    }

    function recalculateTotal() {
      const c = updateRowState(cashInput, diffCash, badgeCash, expectedCash);
      const k = updateRowState(cardsInput, diffCards, badgeCards, expectedCards);
      const t = updateRowState(transfersInput, diffTransfers, badgeTransfers, expectedTransfers);

      const sum = c + k + t;
      const diff = sum - totalExpected;

      totalCounted.textContent = formatCurrency(sum);

      if (Math.abs(diff) < 0.01) {
        totalDiff.textContent = '$0.00';
        totalDiff.className = 'text-secondary';
        overallStatus.className = 'inline-flex items-center gap-1 px-2.5 py-0.5 rounded bg-secondary text-on-secondary font-label-sm text-label-sm';
        overallStatus.innerHTML = '<span class="material-symbols-outlined text-[14px]">done_all</span> Balance Cuadrado';
      } else if (diff > 0) {
        totalDiff.textContent = '+' + formatCurrency(diff);
        totalDiff.className = 'text-tertiary';
        overallStatus.className = 'inline-flex items-center gap-1 px-2.5 py-0.5 rounded bg-tertiary text-on-tertiary font-label-sm text-label-sm';
        overallStatus.innerHTML = '<span class="material-symbols-outlined text-[14px]">warning</span> Sobrante detectado';
      } else {
        totalDiff.textContent = '-' + formatCurrency(diff);
        totalDiff.className = 'text-error';
        overallStatus.className = 'inline-flex items-center gap-1 px-2.5 py-0.5 rounded bg-error text-on-error font-label-sm text-label-sm';
        overallStatus.innerHTML = '<span class="material-symbols-outlined text-[14px]">error</span> Faltante detectado';
      }
    }

    [cashInput, cardsInput, transfersInput].forEach(inp => {
      inp.addEventListener('input', recalculateTotal);
    });

    document.getElementById('btn-quick-fill')?.addEventListener('click', () => {
      cashInput.value = expectedCash.toFixed(2);
      cardsInput.value = expectedCards.toFixed(2);
      transfersInput.value = expectedTransfers.toFixed(2);
      recalculateTotal();
    });

    function executeClosure() {
      const confirmed = confirm('¿Desea cerrar definitivamente el Turno Mañana #042 e imprimir el Comprobante Fiscal Z?');
      if (confirmed) {
        alert('Cierre de caja completado exitosamente. Se ha enviado la orden de impresión térmica del corte Z.');
      }
    }

    document.getElementById('btn-close-shift')?.addEventListener('click', executeClosure);

    document.addEventListener('keydown', (e) => {
      if (e.key === 'F10') {
        e.preventDefault();
        executeClosure();
      }
    });

    document.getElementById('btn-report-x')?.addEventListener('click', () => {
      alert('Generando vista previa de Reporte X (Corte Parcial sin cierre)...');
    });

    document.getElementById('btn-pause')?.addEventListener('click', () => {
      alert('Arqueo pausado. El turno permanece activo en el sistema.');
    });
  })();
</script>
