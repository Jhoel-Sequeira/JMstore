<script>
    (function initBrandHubManagement() {
      // Elements
      const searchInput = document.getElementById('brandSearchInput');
      const clearSearchBtn = document.getElementById('clearSearchBtn');
      const cards = document.querySelectorAll('.brand-card');
      const filterTabs = document.querySelectorAll('.filter-tab');
      const brandCountLabel = document.getElementById('brandCountLabel');
      
      const btnGridView = document.getElementById('btnGridView');
      const btnTableView = document.getElementById('btnTableView');
      const brandsGrid = document.getElementById('brandsGrid');
      const brandsTableWrapper = document.getElementById('brandsTableWrapper');

      const modal = document.getElementById('newBrandModal');
      const openModalBtn = document.getElementById('openNewBrandModal');
      const closeModalBtn = document.getElementById('closeNewBrandModal');
      const cancelModalBtn = document.getElementById('cancelModalBtn');
      const addBrandForm = document.getElementById('addBrandForm');
      const copyInviteBtn = document.getElementById('copyInviteLink');
      const copyLinkText = document.getElementById('copyLinkText');

      let currentFilter = 'all';

      // Filtering and Searching handler
      function applyFilterAndSearch() {
        const query = (searchInput.value || '').trim().toLowerCase();
        let visibleCount = 0;

        cards.forEach(card => {
          const name = (card.getAttribute('data-name') || '').toLowerCase();
          const contact = (card.getAttribute('data-contact') || '').toLowerCase();
          const status = card.getAttribute('data-status');

          const matchesQuery = !query || name.includes(query) || contact.includes(query);
          const matchesFilter = (currentFilter === 'all') || (status === currentFilter);

          if (matchesQuery && matchesFilter) {
            card.classList.remove('hidden');
            visibleCount++;
          } else {
            card.classList.add('hidden');
          }
        });

        brandCountLabel.textContent = visibleCount;
        clearSearchBtn.classList.toggle('hidden', query.length === 0);
      }

      // Filter Pill clicks
      filterTabs.forEach(tab => {
        tab.addEventListener('click', () => {
          filterTabs.forEach(t => {
            t.classList.remove('bg-primary', 'text-on-primary');
            t.classList.add('bg-surface-container-low', 'text-on-surface-variant');
          });
          tab.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
          tab.classList.add('bg-primary', 'text-on-primary');

          currentFilter = tab.getAttribute('data-filter');
          applyFilterAndSearch();
        });
      });

      // Search typing
      searchInput.addEventListener('input', applyFilterAndSearch);

      // Clear search
      clearSearchBtn.addEventListener('click', () => {
        searchInput.value = '';
        applyFilterAndSearch();
        searchInput.focus();
      });

      // View Switcher (Grid vs Table)
      btnGridView.addEventListener('click', () => {
        brandsGrid.classList.remove('hidden');
        brandsTableWrapper.classList.add('hidden');
        btnGridView.classList.add('bg-surface-container-lowest', 'text-primary', 'shadow-xs');
        btnGridView.classList.remove('text-on-surface-variant');
        btnTableView.classList.remove('bg-surface-container-lowest', 'text-primary', 'shadow-xs');
        btnTableView.classList.add('text-on-surface-variant');
      });

      btnTableView.addEventListener('click', () => {
        brandsGrid.classList.add('hidden');
        brandsTableWrapper.classList.remove('hidden');
        btnTableView.classList.add('bg-surface-container-lowest', 'text-primary', 'shadow-xs');
        btnTableView.classList.remove('text-on-surface-variant');
        btnGridView.classList.remove('bg-surface-container-lowest', 'text-primary', 'shadow-xs');
        btnGridView.classList.add('text-on-surface-variant');
      });

      // Modal Controls
      function openModal() {
        modal.classList.remove('hidden');
        modal.classList.add('flex');
      }
      function closeModal() {
        modal.classList.add('hidden');
        modal.classList.remove('flex');
      }

      openModalBtn.addEventListener('click', openModal);
      closeModalBtn.addEventListener('click', closeModal);
      cancelModalBtn.addEventListener('click', closeModal);
      modal.addEventListener('click', (e) => {
        if (e.target === modal) closeModal();
      });

      // Form Submit Toast/Simulation
      addBrandForm.addEventListener('submit', (e) => {
        e.preventDefault();
        closeModal();
        alert('Marca agregada al registro preliminar. Contrato modelo enviado al contacto.');
        addBrandForm.reset();
      });

      // Copy Onboarding Link Micro-interaction
      copyInviteBtn.addEventListener('click', () => {
        const dummyUrl = 'https://pos.brandhub.io/join/partner?token=bh_8f912e';
        navigator.clipboard?.writeText(dummyUrl).then(() => {
          copyLinkText.textContent = '¡Enlace Copiado al Portapapeles!';
          setTimeout(() => {
            copyLinkText.textContent = 'Copiar Enlace de Registro';
          }, 2400);
        }).catch(() => {
          copyLinkText.textContent = '¡Enlace Copiado!';
          setTimeout(() => {
            copyLinkText.textContent = 'Copiar Enlace de Registro';
          }, 2400);
        });
      });
    })();
  </script>
