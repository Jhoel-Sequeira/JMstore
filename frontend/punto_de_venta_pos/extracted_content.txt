<script>
  function flashAddToCart(btn) {
    const originalText = btn.innerHTML;
    btn.innerHTML = '<span class="material-symbols-outlined text-[18px] text-secondary">check</span><span>Agregado</span>';
    btn.classList.add('bg-secondary-fixed', 'text-on-secondary-fixed-variant');
    setTimeout(() => {
      btn.innerHTML = originalText;
      btn.classList.remove('bg-secondary-fixed', 'text-on-secondary-fixed-variant');
    }, 900);
  }

  function selectPayment(selectedBtn) {
    document.querySelectorAll('.pay-btn').forEach(btn => {
      btn.classList.remove('active', 'bg-primary', 'text-on-primary');
      btn.classList.add('bg-surface-container', 'text-on-surface');
    });
    selectedBtn.classList.add('active', 'bg-primary', 'text-on-primary');
    selectedBtn.classList.remove('bg-surface-container', 'text-on-surface');
  }

  function processCheckout() {
    const modal = document.getElementById('successModal');
    modal.classList.remove('hidden');
  }

  function closeModal() {
    const modal = document.getElementById('successModal');
    modal.classList.add('hidden');
    const input = document.getElementById('scannerInput');
    if (input) input.focus();
  }

  document.querySelectorAll('.brand-filter-pill').forEach(pill => {
    pill.addEventListener('click', () => {
      document.querySelectorAll('.brand-filter-pill').forEach(p => {
        p.classList.remove('active', 'bg-primary', 'text-on-primary');
        p.classList.add('bg-surface-container', 'text-on-surface-variant');
      });
      pill.classList.add('active', 'bg-primary', 'text-on-primary');
      pill.classList.remove('bg-surface-container', 'text-on-surface-variant');
    });
  });

  document.addEventListener('keydown', (e) => {
    if (e.key === 'F2') {
      e.preventDefault();
      const input = document.getElementById('scannerInput');
      if (input) input.focus();
    }
    if (e.key === 'F12') {
      e.preventDefault();
      processCheckout();
    }
    if (e.key === 'Enter') {
      const modal = document.getElementById('successModal');
      if (!modal.classList.contains('hidden')) {
        closeModal();
      }
    }
  });
</script>
