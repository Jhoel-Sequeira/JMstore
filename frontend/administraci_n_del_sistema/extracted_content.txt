<script>
    // Micro-interaction handlers for User Management Screen
    document.addEventListener('DOMContentLoaded', () => {
      const openModalBtn = document.getElementById('openInviteModalBtn');
      const closeModalBtn = document.getElementById('closeInviteModalBtn');
      const cancelModalBtn = document.getElementById('cancelInviteBtn');
      const confirmInviteBtn = document.getElementById('confirmInviteBtn');
      const modal = document.getElementById('inviteModal');
      const modalCard = document.getElementById('inviteModalCard');
      const searchInput = document.getElementById('userSearchInput');

      function openModal() {
        modal.classList.remove('opacity-0', 'pointer-events-none');
        modalCard.classList.remove('scale-95');
        modalCard.classList.add('scale-100');
      }

      function closeModal() {
        modal.classList.add('opacity-0', 'pointer-events-none');
        modalCard.classList.remove('scale-100');
        modalCard.classList.add('scale-95');
      }

      if (openModalBtn) openModalBtn.addEventListener('click', openModal);
      if (closeModalBtn) closeModalBtn.addEventListener('click', closeModal);
      if (cancelModalBtn) cancelModalBtn.addEventListener('click', closeModal);
      if (confirmInviteBtn) {
        confirmInviteBtn.addEventListener('click', () => {
          confirmInviteBtn.innerHTML = '<span class="material-symbols-outlined text-[16px] animate-spin">refresh</span> Enviando...';
          setTimeout(() => {
            confirmInviteBtn.innerHTML = 'Enviar Invitación';
            closeModal();
          }, 600);
        });
      }

      // Close modal on backdrop click
      modal.addEventListener('click', (e) => {
        if (e.target === modal) closeModal();
      });

      // Simple Table Filter Interaction
      const roleFilter = document.getElementById('roleFilterSelect');
      const rows = document.querySelectorAll('tbody tr');

      function filterTable() {
        const query = searchInput.value.toLowerCase();
        const role = roleFilter.value;

        rows.forEach(row => {
          const text = row.innerText.toLowerCase();
          const matchesQuery = text.includes(query);
          let matchesRole = true;

          if (role === 'admin') matchesRole = text.includes('administrador');
          if (role === 'cashier') matchesRole = text.includes('cajero');
          if (role === 'brand') matchesRole = text.includes('gestor de marca');

          if (matchesQuery && matchesRole) {
            row.style.display = '';
          } else {
            row.style.display = 'none';
          }
        });
      }

      if (searchInput) searchInput.addEventListener('input', filterTable);
      if (roleFilter) roleFilter.addEventListener('change', filterTable);
    });
  </script>
